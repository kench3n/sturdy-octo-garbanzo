Example 1: if

    let age = 20;

    if (age >= 18) {
        console.log("You can vote.");
    }

Output: You can vote.

Example 2: if/else

    let temperature = 45;

    if (temperature > 60) {
        console.log("Wear a t-shirt.");
    } else {
        console.log("Bring a jacket.");
    }

Output: Bring a jacket.

Example 3: else if

    let score = 84;

    if (score >= 90) {
        console.log("A");
    } else if (score >= 80) {
        console.log("B");
    } else {
        console.log("C or lower");
    }

Output: B
